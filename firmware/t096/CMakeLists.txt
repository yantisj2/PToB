cmake_minimum_required(VERSION 3.20.0)
list(APPEND BOARD_ROOT ${CMAKE_CURRENT_SOURCE_DIR})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
# One RC7 application, two runtime roles (GM and RC). USB export is standard.
if(NOT R2_DC_PROFILE AND NOT R2_PC_PROFILE)
 set(R2_RC_PROFILE ON)
endif()
set(R2_USB_SOURCE_STREAM ON)
set(R2_USB_SOF_PROBE ON)
set(R2_USB_REFERENCE_PROBE ON)
project(ptob_t096 VERSION 0.15.145)
option(R2_CONTROLLER_DIAGNOSTICS "Bounded controller diagnostic canary" OFF)
if(R2_CONTROLLER_DIAGNOSTICS AND (R2_DC_PROFILE OR R2_PC_PROFILE))
 message(FATAL_ERROR "Controller diagnostic canary requires normal GM/RC")
endif()
if(R2_DC_PROFILE AND R2_PC_PROFILE)
 message(FATAL_ERROR "DC and PC are separate images")
endif()
if(NOT DEFINED R2_LORA_TRANSMITTER_FACTORY)
 set(R2_LORA_TRANSMITTER_FACTORY "0000000000000000")
endif()
string(LENGTH "${R2_LORA_TRANSMITTER_FACTORY}" R2_LORA_FACTORY_LENGTH)
if(NOT R2_LORA_FACTORY_LENGTH EQUAL 16 OR NOT R2_LORA_TRANSMITTER_FACTORY MATCHES "^[0-9a-f]+$")
 message(FATAL_ERROR "LoRa transmitter factory must be exactly16 lowercase hex digits")
endif()
if(R2_PC_PROFILE)
 include(cmake/power_control.cmake)
 return()
endif()
if(R2_DC_PROFILE)
 include(cmake/direct_connect.cmake)
 return()
endif()
if(R2_USB_SOURCE_STREAM)
 if(NOT R2_USB_SOF_PROBE)
  message(FATAL_ERROR "USB source evidence requires hardware SOF profile")
 endif()
 target_compile_definitions(app PRIVATE R2_USB_SOURCE_STREAM=1)
 target_sources(app PRIVATE src/usb_source.c)
endif()
if(R2_USB_REFERENCE_PROBE)
 if(NOT R2_RC_PROFILE)
  message(FATAL_ERROR "USB API probe requires the separately reviewed RC profile")
 endif()
 target_sources(app PRIVATE src/usb_reference_interface.c)
endif()
if(R2_USB_SOF_PROBE)
 if(NOT R2_USB_REFERENCE_PROBE)
  message(FATAL_ERROR "Raw SOF probe requires the reviewed USB reference interface")
 endif()
 target_compile_definitions(app PRIVATE R2_USB_SOF_PROBE=1)
 target_sources(app PRIVATE src/usb_sof_probe.c src/usb_sof_observer.c)
endif()
foreach(pair R2_SOURCE_SHA R2_CHILD_COMMIT)
  if(NOT ${pair} MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Use v12_r2/tools/build.py for identity-bound builds")
  endif()
endforeach()
if(NOT EXISTS "${R2_ENROLLMENT_HEADER}")
  message(FATAL_ERROR "Local enrollment header required; never include secrets in public source")
endif()
file(SHA256 "${R2_ENROLLMENT_HEADER}" R2_PRIVATE_CONFIG_SHA)
if(NOT R2_LORA_TRANSMITTER_FACTORY STREQUAL "0000000000000000")
 string(SHA256 R2_PRIVATE_CONFIG_SHA "${R2_PRIVATE_CONFIG_SHA}:lora-tx:${R2_LORA_TRANSMITTER_FACTORY}")
endif()
file(SHA256 "${PROJECT_BINARY_DIR}/zephyr/.config" R2_KERNEL_CONFIG_SHA)
if(R2_CONTROLLER_DIAGNOSTICS)
 string(SHA256 R2_PRIVATE_CONFIG_SHA "${R2_PRIVATE_CONFIG_SHA}:controller-diag:1")
endif()
string(SHA256 R2_WATCHDOG_BINDING "${R2_SOURCE_SHA}:${R2_PRIVATE_CONFIG_SHA}:${R2_KERNEL_CONFIG_SHA}")
set(R2_CONTROLLER_DIAGNOSTICS_ENABLED ${R2_CONTROLLER_DIAGNOSTICS})
configure_file(include/build_identity.h.in ${CMAKE_CURRENT_BINARY_DIR}/build_identity.h @ONLY)
target_include_directories(app PRIVATE include ${CMAKE_CURRENT_BINARY_DIR})
target_sources(app PRIVATE src/main.c src/board.c src/console.c src/core.c src/platform.c src/clock.c src/nmea.c src/time_service.c src/gnss.c src/ui_render.c src/ui_service.c src/display.c src/timing_events.c src/initiator_diag.c)
target_sources(app PRIVATE src/precision.c src/capture.c src/radio_capture.c src/radio_audit.c src/rc_boundary.c src/rc_shadow.c protocol/precision_codec.c)
if(R2_RC_AUTO)
 if(NOT R2_SWITCHABLE_FIELD)
  target_compile_definitions(app PRIVATE R2_RC_AUTO=1)
 endif()
 target_sources(app PRIVATE src/rc_slots.c)
endif()
if(R2_RC_PROFILE)
 if(R2_BOUNDARY_FIELD OR R2_SWITCHABLE_FIELD OR R2_RC_AUTO)
  message(FATAL_ERROR "RC7 profile cannot mix legacy RC diagnostic owners")
 endif()
 target_compile_definitions(app PRIVATE R2_RC_PROFILE=1 R2_SWITCHABLE_FIELD=1)
 if(CONFIG_ARM AND NOT CONFIG_BOARD_NRF52_BSIM)
  target_compile_definitions(app PRIVATE R2_WATCHDOG=1)
  target_sources(app PRIVATE src/watchdog.c src/watchdog_policy.c)
 endif()
 set_source_files_properties(src/ble_service.c PROPERTIES COMPILE_DEFINITIONS R2_NORMAL_BACKEND=1)
 set_source_files_properties(src/rc_dynamic_service.c PROPERTIES COMPILE_DEFINITIONS "R2_DYNAMIC_BACKEND=1;R2_RC_DYNAMIC=1")
 set_source_files_properties(src/rc_shadow.c PROPERTIES COMPILE_DEFINITIONS R2_SHADOW_FIELD_MAP=1)
 target_sources(app PRIVATE src/rc_profile_service.c src/rc_dynamic_service.c src/rc_slots.c protocol/rc_dynamic.c)
 target_sources(app PRIVATE src/rc_unicast.c protocol/unicast_time.c)
 zephyr_link_libraries(-Wl,--undefined=r2_uc_contract)
 target_sources(app PRIVATE src/radio_precision.c src/radio_service.c src/relay_precision.c src/relay_service.c protocol/relay_codec.c)
elseif(R2_BOUNDARY_FIELD)
 if(NOT R2_SWITCHABLE_FIELD)
  target_compile_definitions(app PRIVATE R2_BOUNDARY_FIELD=1)
 endif()
 target_sources(app PRIVATE src/rc_field_service.c src/rc_gm_adapter.c protocol/relay_codec.c)
 if(R2_SWITCHABLE_FIELD)
  target_compile_definitions(app PRIVATE R2_SWITCHABLE_FIELD=1)
  set_source_files_properties(src/ble_service.c PROPERTIES COMPILE_DEFINITIONS R2_NORMAL_BACKEND=1)
  set_source_files_properties(src/rc_field_service.c PROPERTIES COMPILE_DEFINITIONS "R2_FIELD_BACKEND=1;R2_BOUNDARY_FIELD=1;R2_RC_AUTO=1")
  set_source_files_properties(src/rc_boundary.c src/rc_gm_adapter.c src/rc_slots.c PROPERTIES COMPILE_DEFINITIONS R2_RC_AUTO=1)
  set_source_files_properties(src/rc_shadow.c PROPERTIES COMPILE_DEFINITIONS R2_SHADOW_FIELD_MAP=1)
  target_sources(app PRIVATE src/service_switch.c src/ble_service.c src/radio_precision.c src/radio_service.c src/relay_precision.c src/relay_service.c protocol/network.c protocol/ble_transport.c protocol/authority_codec.c src/group_admission.c)
 endif()
else()
 target_sources(app PRIVATE src/radio_precision.c src/radio_service.c src/relay_precision.c src/relay_service.c protocol/relay_codec.c)
endif()
target_include_directories(app PRIVATE ${ZEPHYR_BASE}/subsys/bluetooth/controller/ll_sw/nordic/hal/nrf5/radio)
# Exact pinned clock provenance correction and merged upstream ticker teardown fix.
include(cmake/usb_ep0_recovery.cmake)
include(cmake/clock_handoff.cmake)
include(cmake/ticker_backport.cmake)
include(cmake/radio_capture.cmake)
include(cmake/precision_schedule.cmake)
include(cmake/controller_diag.cmake)
target_compile_options(app PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)

target_compile_definitions(app PRIVATE R2_NATIVE_BLE=1)
if(R2_RC_PROFILE AND NOT R2_PC_PROFILE)
 target_compile_definitions(app PRIVATE R2_GM5_NATIVE=1)
endif()
target_include_directories(app PRIVATE protocol)
target_sources(app PRIVATE src/diagnostic.c src/health.c protocol/election_policy.c)
# RC8 is a separate secondary estimate. The enrolled factory binding gates startup.
target_compile_definitions(app PRIVATE R2_LORA=1)
target_sources(app PRIVATE protocol/lora_rc8.c src/lora_rc8_policy.c src/lora_source.c src/lora_service.c src/lora_hw.c)
if(NOT R2_BOUNDARY_FIELD)
 target_sources(app PRIVATE src/ble_service.c protocol/network.c protocol/ble_transport.c protocol/authority_codec.c)
endif()

configure_file(${R2_ENROLLMENT_HEADER} ${CMAKE_CURRENT_BINARY_DIR}/r2_enrollment.h COPYONLY)

target_sources(app PRIVATE protocol/gm_diagnostics.c protocol/group_join.c protocol/group_auth.c protocol/group_control.c protocol/group_precision.c protocol/role_announcement.c)

if(NOT R2_BOUNDARY_FIELD)
 target_sources(app PRIVATE src/group_admission.c)
endif()

# Generated from the shared registry, display-only.
target_compile_definitions(app PRIVATE PTOB_NAME_REGISTRY=1)

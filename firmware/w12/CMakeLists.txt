cmake_minimum_required(VERSION 3.16)
# Public W12 successor: one RC receiver profile. No legacy bench roster or GM.
set(IDF_TARGET esp32s3)
set(PROJECT_VER "0.2.5")
set(COMPONENTS main)
set(V12_R2_PROFILE "r2-g1")
set(V12_UNIT "W12")
set(V12_R2_FIELD_CLIENT ON)
set(V12_R2_RX_SHADOW ON)
set(V12_R2_RC7_EXCLUDE_DIRECT_GM OFF)
set(V12_R2_OUTPUT_CORRECTION_NS "0")
set(V12_R2_OUTPUT_ROOT "0")
set(V12_R2_OUTPUT_BOOT "0")
set(V12_R2_OUTPUT_SOURCE_EPOCH "0")
set(V12_R2_G1_LISTENING_PERCENT "100" CACHE STRING "Boot listening percent: 50 or 100")
option(V12_R2_RC7_TX_ESTIMATE_LAB "Experimental RC relay with unknown TX bound" OFF)
option(PTOB_RC9 "Separate RC8-derived RC9 BLE service" OFF)
option(PTOB_W12_LORA_RX "Passive LR2021 RC8 fallback" OFF)
option(PTOB_W12_FINE_CLOCK "Experimental W12 GPIO16 MCPWM1 fine clock" OFF)
option(PTOB_W12_GNSS_INDICATORS "Explicit Meshnology W12 L76K RX39/PPS41 indicator input profile" OFF)
option(PTOB_W12_GNSS_FAST_MAP_DIAGNOSTIC "Diagnostic-only fast software counter-map bracket" OFF)
if(PTOB_W12_GNSS_FAST_MAP_DIAGNOSTIC AND (NOT PTOB_W12_GNSS_INDICATORS OR NOT V12_R2_RX_SHADOW OR NOT V12_R2_PROFILE STREQUAL "r2-g1"))
 message(FATAL_ERROR "Fast map diagnostic requires explicit W12 GNSS G1 and pinned RX-shadow driver ABI")
endif()
set(V12_R2_TX_SHADOW ${V12_R2_RC7_TX_ESTIMATE_LAB})
if(NOT V12_R2_G1_LISTENING_PERCENT MATCHES "^(50|100)$")
 message(FATAL_ERROR "Listening must be 50 or 100")
endif()
if(V12_R2_RC7_TX_ESTIMATE_LAB AND NOT V12_R2_G1_LISTENING_PERCENT STREQUAL "100")
 message(FATAL_ERROR "Estimated relay requires BLE100; BLE50 remains client-only")
endif()
if(PTOB_W12_FINE_CLOCK AND (PTOB_W12_GNSS_FAST_MAP_DIAGNOSTIC OR NOT V12_R2_G1_LISTENING_PERCENT STREQUAL "100"))
 message(FATAL_ERROR "Fine clock requires BLE100 and the ordinary GNSS map bracket")
endif()
if(PTOB_RC9 AND (NOT PTOB_W12_LORA_RX OR NOT V12_R2_RC7_TX_ESTIMATE_LAB))
 message(FATAL_ERROR "RC9 W12 service requires LR2021 RX and reviewed BLE100 relay owner")
endif()
foreach(required V12_R2_ENROLLMENT_HEADER PTOB_DEVICE_HEADER PTOB_REGISTRY_HEADER PTOB_FINE_HEADER PTOB_LORA_HEADER)
 if(NOT EXISTS "${${required}}")
  message(FATAL_ERROR "Generated local enrollment/device/registry header required")
 endif()
endforeach()
string(LENGTH "${V12_SOURCE_REVISION}" source_revision_length)
if(NOT V12_SOURCE_REVISION MATCHES "^[0-9a-f]+$" OR NOT source_revision_length EQUAL 40)
 message(FATAL_ERROR "Exact reviewed public source commit required")
endif()
file(SHA256 "${V12_R2_ENROLLMENT_HEADER}" r2_enrollment_sha)
file(SHA256 "$ENV{IDF_PATH}/components/bt/controller/lib_esp32c3_family/esp32s3/libbtdm_app.a" controller_sha)
if(NOT controller_sha STREQUAL "513488f088f7806c5a815580fd00e189265962a8824091b1e60618b7f345d0c1")
 message(FATAL_ERROR "Pinned S3 controller changed")
endif()
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.g1.defaults")
if(V12_R2_RC7_TX_ESTIMATE_LAB)
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.rc7-server.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ptob_w12)
file(SHA256 "${SDKCONFIG}" effective_config_sha)
file(SHA256 "${PTOB_DEVICE_HEADER}" device_sha)
file(SHA256 "${PTOB_REGISTRY_HEADER}" registry_sha)
file(SHA256 "${PTOB_FINE_HEADER}" fine_sha)
file(SHA256 "${PTOB_LORA_HEADER}" lora_sha)
string(SHA256 native_build_id "${V12_SOURCE_REVISION}|${effective_config_sha}|${controller_sha}|${r2_enrollment_sha}|${device_sha}|${registry_sha}|${V12_R2_G1_LISTENING_PERCENT}|${V12_R2_RC7_TX_ESTIMATE_LAB}|${PTOB_W12_GNSS_FAST_MAP_DIAGNOSTIC}|${PTOB_W12_FINE_CLOCK}|${fine_sha}|${lora_sha}|${PTOB_RC9}|PTOB_W12_${PROJECT_VER}")
target_compile_definitions(__idf_main PRIVATE V12_BUILD_ID="${native_build_id}")
file(WRITE "${CMAKE_BINARY_DIR}/v12_build_identity.json" "{\n  \"source_commit\": \"${V12_SOURCE_REVISION}\",\n  \"version\": \"${PROJECT_VER}\",\n  \"build_id\": \"${native_build_id}\",\n  \"sdkconfig_sha256\": \"${effective_config_sha}\",\n  \"controller_sha256\": \"${controller_sha}\",\n  \"r2_enrollment_sha256\": \"${r2_enrollment_sha}\",\n  \"device_header_sha256\": \"${device_sha}\",\n  \"registry_header_sha256\": \"${registry_sha}\"\n}\n")

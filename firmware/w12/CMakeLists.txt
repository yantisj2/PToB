cmake_minimum_required(VERSION 3.16)
# Public W12 successor: one RC receiver profile. No legacy bench roster or GM.
set(IDF_TARGET esp32s3)
set(PROJECT_VER "0.1.2")
set(COMPONENTS main)
set(V12_R2_PROFILE "r2-g1")
set(V12_UNIT "W12")
set(V12_R2_FIELD_CLIENT ON)
set(V12_R2_RX_SHADOW ON)
set(V12_R2_RC7_EXCLUDE_DIRECT_GM OFF)
set(V12_R2_OUTPUT_CORRECTION_NS "0")
set(V12_R2_OUTPUT_ROOT "0")
set(V12_R2_OUTPUT_BOOT "0")
set(V12_R2_OUTPUT_SOURCE_EPOCH "0")
set(V12_R2_G1_LISTENING_PERCENT "100" CACHE STRING "Boot listening percent: 50 or 100")
option(V12_R2_RC7_TX_ESTIMATE_LAB "Experimental RC relay with unknown TX bound" OFF)
set(V12_R2_TX_SHADOW ${V12_R2_RC7_TX_ESTIMATE_LAB})
if(NOT V12_R2_G1_LISTENING_PERCENT MATCHES "^(50|100)$")
 message(FATAL_ERROR "Listening must be 50 or 100")
endif()
if(V12_R2_RC7_TX_ESTIMATE_LAB AND NOT V12_R2_G1_LISTENING_PERCENT STREQUAL "100")
 message(FATAL_ERROR "Estimated relay requires BLE100; BLE50 remains client-only")
endif()
foreach(required V12_R2_ENROLLMENT_HEADER PTOB_DEVICE_HEADER PTOB_REGISTRY_HEADER)
 if(NOT EXISTS "${${required}}")
  message(FATAL_ERROR "Generated local enrollment/device/registry header required")
 endif()
endforeach()
string(LENGTH "${V12_SOURCE_REVISION}" source_revision_length)
if(NOT V12_SOURCE_REVISION MATCHES "^[0-9a-f]+$" OR NOT source_revision_length EQUAL 40)
 message(FATAL_ERROR "Exact reviewed public source commit required")
endif()
file(SHA256 "${V12_R2_ENROLLMENT_HEADER}" r2_enrollment_sha)
file(SHA256 "$ENV{IDF_PATH}/components/bt/controller/lib_esp32c3_family/esp32s3/libbtdm_app.a" controller_sha)
if(NOT controller_sha STREQUAL "513488f088f7806c5a815580fd00e189265962a8824091b1e60618b7f345d0c1")
 message(FATAL_ERROR "Pinned S3 controller changed")
endif()
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.g1.defaults")
if(V12_R2_RC7_TX_ESTIMATE_LAB)
 list(APPEND SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.rc7-server.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(ptob_w12)
file(SHA256 "${SDKCONFIG}" effective_config_sha)
file(SHA256 "${PTOB_DEVICE_HEADER}" device_sha)
file(SHA256 "${PTOB_REGISTRY_HEADER}" registry_sha)
string(SHA256 native_build_id "${V12_SOURCE_REVISION}|${effective_config_sha}|${controller_sha}|${r2_enrollment_sha}|${device_sha}|${registry_sha}|${V12_R2_G1_LISTENING_PERCENT}|${V12_R2_RC7_TX_ESTIMATE_LAB}|PTOB_W12_0.1.2")
target_compile_definitions(__idf_main PRIVATE V12_BUILD_ID="${native_build_id}")
file(WRITE "${CMAKE_BINARY_DIR}/v12_build_identity.json" "{\n  \"source_commit\": \"${V12_SOURCE_REVISION}\",\n  \"version\": \"0.1.2\",\n  \"build_id\": \"${native_build_id}\",\n  \"sdkconfig_sha256\": \"${effective_config_sha}\",\n  \"controller_sha256\": \"${controller_sha}\",\n  \"r2_enrollment_sha256\": \"${r2_enrollment_sha}\",\n  \"device_header_sha256\": \"${device_sha}\",\n  \"registry_header_sha256\": \"${registry_sha}\"\n}\n")

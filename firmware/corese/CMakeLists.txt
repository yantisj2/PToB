cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET esp32s3)
set(PROJECT_VER "0.1.15")
set(COMPONENTS main)
if(NOT EXISTS "${CORESE_DEPENDENCIES}/M5GFX/CMakeLists.txt" OR
   NOT EXISTS "${CORESE_DEPENDENCIES}/M5Unified/CMakeLists.txt")
 message(FATAL_ERROR "Pinned public M5GFX/M5Unified dependency directory required")
endif()
set(EXTRA_COMPONENT_DIRS "${CORESE_DEPENDENCIES}/M5GFX" "${CORESE_DEPENDENCIES}/M5Unified")
if(CORESE_LIVE)
 set(CORESE_APP "${CMAKE_CURRENT_LIST_DIR}")
 set(V12_R2_PROFILE "r2-g1")
 set(V12_UNIT "CORESE")
 set(V12_R2_FIELD_CLIENT ON)
 set(V12_R2_RX_SHADOW ON)
 set(V12_R2_RC7_TX_ESTIMATE_LAB OFF)
 set(V12_R2_TX_SHADOW OFF)
 set(V12_R2_RC7_EXCLUDE_DIRECT_GM OFF)
 set(V12_R2_OUTPUT_CORRECTION_NS 0)
 set(V12_R2_OUTPUT_ROOT 0)
 set(V12_R2_OUTPUT_BOOT 0)
 set(V12_R2_OUTPUT_SOURCE_EPOCH 0)
 set(V12_R2_G1_LISTENING_PERCENT 100)
 set(PTOB_W12_GNSS_INDICATORS OFF)
 set(PTOB_W12_GNSS_FAST_MAP_DIAGNOSTIC OFF)
 foreach(required V12_R2_ENROLLMENT_HEADER PTOB_DEVICE_HEADER PTOB_REGISTRY_HEADER PTOB_LORA_HEADER PTOB_FINE_HEADER)
  if(NOT EXISTS "${${required}}")
   message(FATAL_ERROR "Local enrolled exact-device input required: ${required}")
  endif()
 endforeach()
 file(SHA256 "${V12_R2_ENROLLMENT_HEADER}" r2_enrollment_sha)
 file(SHA256 "$ENV{IDF_PATH}/components/bt/controller/lib_esp32c3_family/esp32s3/libbtdm_app.a" controller_sha)
 if(NOT controller_sha STREQUAL "513488f088f7806c5a815580fd00e189265962a8824091b1e60618b7f345d0c1")
  message(FATAL_ERROR "Pinned controller changed")
 endif()
 set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/../n30/sdkconfig.defaults;${CMAKE_CURRENT_LIST_DIR}/../n30/sdkconfig.g1.defaults;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.live.defaults")
else()
 set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(CORESE_LIVE)
 project(ptob_corese)
 file(SHA256 "${SDKCONFIG}" effective_config_sha)
 foreach(pair "${PTOB_DEVICE_HEADER}|device_sha" "${PTOB_REGISTRY_HEADER}|registry_sha" "${PTOB_FINE_HEADER}|fine_sha")
  string(REPLACE "|" ";" fields "${pair}")
  list(GET fields 0 source)
  list(GET fields 1 dest)
  file(SHA256 "${source}" ${dest})
 endforeach()
 string(SHA256 native_build_id "${V12_SOURCE_REVISION}|${CORESE_SOURCE_SHA}|${effective_config_sha}|${controller_sha}|${r2_enrollment_sha}|${device_sha}|${registry_sha}|${fine_sha}|PTOB_CORESE_${PROJECT_VER}")
 target_compile_definitions(__idf_main PRIVATE V12_BUILD_ID="${native_build_id}")
 file(WRITE "${CMAKE_BINARY_DIR}/v12_build_identity.json" "{\n \"version\":\"${PROJECT_VER}\",\n \"build_id\":\"${native_build_id}\",\n \"source_commit\":\"${V12_SOURCE_REVISION}\",\n \"source_sha256\":\"${CORESE_SOURCE_SHA}\",\n \"sdkconfig_sha256\":\"${effective_config_sha}\",\n \"controller_sha256\":\"${controller_sha}\",\n \"r2_enrollment_sha256\":\"${r2_enrollment_sha}\",\n \"device_header_sha256\":\"${device_sha}\",\n \"registry_header_sha256\":\"${registry_sha}\",\n \"fine_header_sha256\":\"${fine_sha}\"\n}\n")
else()
 project(ptob_corese_ui)
endif()
